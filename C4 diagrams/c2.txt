workspace "ConnectSphere" "C4 model of the ConnectSphere platform: system context, containers, and the components of the Flask application. CFnn and W70n refer to the functional requirements and Week 7 changes in the ConnectSphere functional requirements document." {

    model {
        # System Users / Actors: the seven roles (CF01). Accounts are pre-loaded; there is no self-registration in Release 1.
        organiser = person "Event Organiser" "Drafts and submits event requests, answers clarification questions, resubmits rejected requests, and requests changes to approved events through the coordinator."
        lead = person "Event Coordinator Lead" "Reviews the unassigned queue, assigns and reassigns coordinators, and oversees the coordinators and active events they assigned."
        coordinator = person "Event Coordinator" "Reviews assigned requests, plans venues and equipment, edits event details, decides change requests, sets up registration, and confirms events."
        venueStaff = person "Venue Staff" "Maintains the venue catalogue, setup and turnaround times and unavailable periods; decides booking requests and manages tentative holds."
        techStaff = person "Technical Support Staff" "Reviews equipment requirements, checks availability, reserves equipment, and updates technical arrangements."
        safetyOfficer = person "Safety Officer" "Reviews operational safety once venue and equipment are arranged, and approves, rejects or requests changes."
        attendee = person "Attendee" "Browses published events, registers, checks registration status, and withdraws."

        # External Systems
        emailSystem = softwareSystem "Email Gateway / SMTP" "External mail service that delivers notification emails."

        # ConnectSphere System Scope
        connectSphere = softwareSystem "ConnectSphere Platform" "Event planning from request to registration: assignment, review, venue and equipment arrangements, safety approval, confirmation, and attendee registration." {

            # Front-End Layer
            webApp = container "Web Client / UI" "Single-page application with one screen per task, including in-app notifications. The menu shows each role its own screens; every permission is enforced by the API." "React 18 / JavaScript / Vite" "WebClient"

            # Core Execution Layer
            monolithApp = container "ConnectSphere Monolithic Application" "REST API under /api. Checks each caller's stored role and relationship to the event, applies the workflow rules, and saves each change together with its in-app notifications." "Python / Flask / SQLAlchemy" "CoreApp" {
                auth = component "Authentication" "Signs users in and issues the JWT every other API requires. Each request reloads the user, so permissions follow their current role (CF01)." "Flask blueprint: auth"
                eventRequests = component "Event Requests and Drafts" "Saves, lists and resumes an organiser's drafts, validates the compulsory fields, and submits a request into the Lead's unassigned queue (CF02, CF03)." "Flask blueprint: events"
                assignment = component "Coordinator Assignment" "The unassigned queue, assignment and reassignment by a Lead, coordinators' reassignment requests, and the events each Lead supervises (CF05, W705)." "Flask blueprint: events"
                review = component "Request Review" "The assigned coordinator's review list, clarification questions and answers, resubmission, and the recorded approval or rejection (CF04)." "Flask blueprint: events"
                planning = component "Event Planning and Change Requests" "The shared planning view; ordinary edits saved directly and restricted ones sent for review; organiser change requests and the coordinator's decisions, which apply the accepted values (CF07, CF19)." "Flask blueprint: events"
                safetyConfirmation = component "Safety Check and Confirmation" "The Safety Officer's queue and decisions, where a rejection or request for changes returns the event to planning; and the coordinator's confirmation once venues, equipment and safety are approved (CF06, W706)." "Flask blueprint: events"
                equipment = component "Equipment" "Requirements recorded by the coordinator, Technical Support's request list and arrangement updates, availability against stock, overlapping reservations and unavailable items, and reservations that never over-commit stock (CF15, CF16, CF17)." "Flask blueprint: events; resources package"
                venueCatalogue = component "Venue Catalogue" "Venue records that tell a missing value from a recorded absence, setup and turnaround times, change history, and bookings flagged when new times create a conflict (CF08, W701)." "Flask blueprint: venues"
                venueAvailability = component "Venue Availability, Search and Suitability" "The availability calendar, venue search and filtering, and checks of a venue against an event's capacity, layout, accessibility and facilities (CF09, CF10, CF11)." "Flask blueprint: venues; events.suitability"
                venueBookings = component "Venue Bookings and Holds" "Booking requests for one or several venues, Venue Staff approval or rejection, tentative holds with an expiry staff can change, confirming a hold, and cancelling one booking (CF12, CF13, W703, W704)." "Flask blueprint: venues"
                venueDisruption = component "Venue Unavailability and Replacement" "Records unavailable periods, flags the confirmed bookings inside them without cancelling anything, warns coordinators of pending requests and holds there, and handles replacement-venue requests (W702)." "Flask blueprint: venues"
                scheduling = component "Scheduling Rules" "Shared rules for occupied periods (setup, event and turnaround, end not included), overlaps, what blocks a venue, and a hold's current status (CF14, W701)." "Python module: venues.scheduling"
                registrations = component "Registrations" "Registration, status and withdrawal for approved or confirmed events; the coordinator's registration settings; registration lists for the organiser and coordinator (CF18)." "Flask blueprints: registrations, events"
                notifications = component "Notifications" "Lists a user's in-app notifications and marks them read (CF20)." "Flask blueprint: notifications"
                taskClient = component "Background Task Client" "Queues a notification email after the change it reports is saved. A queue failure is logged and never undoes the change." "Celery tasks: tasks.py"
            }
            celeryWorker = container "Celery Worker Process" "Runs background tasks: sends notification emails, and runs the tentative-hold check, which expires holds past their expiry and notifies coordinators 24 hours before and at expiry (W704)." "Python / Celery" "Worker"
            celeryBeat = container "Celery Beat Scheduler" "Queues the tentative-hold check every 5 minutes." "Python / Celery" "Worker"

            # Data & Messaging Layer
            supabaseDb = container "Supabase Database" "Users and roles; events with their review, safety and confirmation decisions; clarifications; reassignment and change requests; venues, their change history and unavailable periods; venue bookings and holds with their conflict and replacement flags; equipment stock, unavailability, requirements and reservations; registrations; in-app notifications." "Managed PostgreSQL / Supabase" "Database"
            redisBroker = container "Redis Queue" "Message broker holding background tasks until a worker takes them." "Redis" "Broker"
        }

        # User Connections to Client
        organiser -> webApp "Drafts, submits and tracks event requests; answers clarifications; requests changes" "HTTP"
        lead -> webApp "Assigns and reassigns coordinators; views supervised events" "HTTP"
        coordinator -> webApp "Reviews requests; plans venues and equipment; decides changes; confirms events" "HTTP"
        venueStaff -> webApp "Maintains venues; decides bookings; manages holds and unavailability" "HTTP"
        techStaff -> webApp "Reviews requirements; checks and reserves equipment" "HTTP"
        safetyOfficer -> webApp "Records safety decisions" "HTTP"
        attendee -> webApp "Browses events; registers and withdraws" "HTTP"

        # Container-level connections, stated before the component ones so they keep these descriptions
        webApp -> monolithApp "Sends API requests to" "HTTP / JSON"
        monolithApp -> supabaseDb "Reads & writes data" "SQL / TCP"
        monolithApp -> redisBroker "Queues notification emails" "Redis Protocol"

        # Web client to the components it calls
        webApp -> auth "Signs in through" "HTTP / JSON"
        webApp -> eventRequests "Saves drafts and submits requests through" "HTTP / JSON"
        webApp -> assignment "Assigns and reassigns coordinators through" "HTTP / JSON"
        webApp -> review "Reviews requests and exchanges clarifications through" "HTTP / JSON"
        webApp -> planning "Edits events and decides change requests through" "HTTP / JSON"
        webApp -> safetyConfirmation "Records safety decisions and confirms events through" "HTTP / JSON"
        webApp -> equipment "Records, checks and reserves equipment through" "HTTP / JSON"
        webApp -> venueCatalogue "Maintains venues through" "HTTP / JSON"
        webApp -> venueAvailability "Views availability, searches and checks suitability through" "HTTP / JSON"
        webApp -> venueBookings "Requests and decides bookings and holds through" "HTTP / JSON"
        webApp -> venueDisruption "Records unavailability and requests replacements through" "HTTP / JSON"
        webApp -> registrations "Registers attendees and manages registration through" "HTTP / JSON"
        webApp -> notifications "Reads notifications through" "HTTP / JSON"

        # Shared rules
        venueAvailability -> scheduling "Works out occupied periods with"
        venueBookings -> scheduling "Rechecks conflicts at decision time with"
        venueDisruption -> scheduling "Finds affected bookings and holds with"
        equipment -> scheduling "Finds overlapping reservations with"

        # Emails, queued after the change and its in-app notification are saved
        assignment -> taskClient "Queues assignment emails through"
        review -> taskClient "Queues decision emails through"
        venueBookings -> taskClient "Queues booking and hold emails through"
        venueDisruption -> taskClient "Queues unavailability emails through"
        taskClient -> redisBroker "Queues email tasks" "Redis Protocol"

        # Persistence: each component saves its records, and notifications with them, in one commit
        auth -> supabaseDb "Reads users" "SQL / TCP"
        eventRequests -> supabaseDb "Reads & writes events" "SQL / TCP"
        assignment -> supabaseDb "Reads & writes assignments, reassignment requests and notifications" "SQL / TCP"
        review -> supabaseDb "Reads & writes decisions, clarifications and notifications" "SQL / TCP"
        planning -> supabaseDb "Reads & writes events, change requests and notifications" "SQL / TCP"
        safetyConfirmation -> supabaseDb "Reads & writes safety decisions, confirmations and notifications" "SQL / TCP"
        equipment -> supabaseDb "Reads & writes requirements, reservations and notifications" "SQL / TCP"
        venueCatalogue -> supabaseDb "Reads & writes venues, their history and conflict flags" "SQL / TCP"
        venueAvailability -> supabaseDb "Reads venues, bookings, holds and unavailable periods" "SQL / TCP"
        venueBookings -> supabaseDb "Reads & writes bookings, holds and notifications" "SQL / TCP"
        venueDisruption -> supabaseDb "Reads & writes unavailable periods, replacement flags and notifications" "SQL / TCP"
        registrations -> supabaseDb "Reads & writes registrations and registration settings" "SQL / TCP"
        notifications -> supabaseDb "Reads notifications and records them as read" "SQL / TCP"

        # Background Processing & Notification Flow
        celeryBeat -> redisBroker "Queues the tentative-hold check" "Redis Protocol"
        redisBroker -> celeryWorker "Delivers tasks to" "Redis Protocol"
        celeryWorker -> supabaseDb "Expires holds and saves hold notifications" "SQL / TCP"
        celeryWorker -> emailSystem "Dispatches emails via" "SMTP"
    }

    views {
        systemContext connectSphere "System_Context" {
            include *
            autolayout tb
        }

        container connectSphere "Containers_View" {
            include *
            autolayout tb
        }

        component monolithApp "Components_View" {
            include *
            autolayout lr
        }

        styles {
            element "Person" {
                shape Person
                background #1b365d
                color #ffffff
            }
            element "Container" {
                background #2b6cb0
                color #ffffff
            }
            element "Component" {
                background #63b3ed
                color #000000
            }
            element "WebClient" {
                shape WebBrowser
                background #319795
                color #ffffff
            }
            element "Database" {
                shape Cylinder
                background #2d3748
                color #ffffff
            }
            element "Broker" {
                shape Pipe
                background #d69e2e
                color #ffffff
            }
            element "Worker" {
                background #4a5568
                color #ffffff
            }
            element "Software System" {
                background #718096
                color #ffffff
            }
        }
    }
}
